nano loop.c

#include <stdio.h>
#include <omp.h>

int main() {
    int sum = 0;

    #pragma omp parallel for reduction(+:sum)
    for (int i = 1; i <= 10; i++) {
        sum += i;
    }

    printf("Sum = %d\n", sum);
    return 0;
}

using private

#include <stdio.h>
#include <omp.h>

int main() {
    int x = 100;

    #pragma omp parallel private(x)
    {
        x = omp_get_thread_num();
        printf("Thread %d: x = %d\n",
               omp_get_thread_num(), x);
    }

    printf("Original x = %d\n", x);
    return 0;
}

firstprivate

#include <stdio.h>
#include <omp.h>

int main() {
    int x = 10;

    #pragma omp parallel firstprivate(x)
    {
        x += omp_get_thread_num();
        printf("Thread %d: x = %d\n",
               omp_get_thread_num(), x);
    }

    printf("Original x = %d\n", x);
    return 0;
}

lastprivate

#include <stdio.h>
#include <omp.h>

int main() {
    int x = 0;

    #pragma omp parallel for lastprivate(x)
    for (int i = 1; i <= 5; i++) {
        x = i;
    }

    printf("Last value of x = %d\n", x);
    return 0;
}

criticalsession

#include <stdio.h>
#include <omp.h>

int main() {
    int count = 0;

    #pragma omp parallel for
    for (int i = 0; i < 1000; i++) {
        #pragma omp critical
        {
            count++;
        }
    }

    printf("Count = %d\n", count);
    return 0;
}
